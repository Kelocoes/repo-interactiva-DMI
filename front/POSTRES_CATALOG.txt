EJEMPLO COMPLETO DE CÓDIGO PARA LA TERMINAL:

 // Colores de mi tienda
 const colorTexto = "#5552F6";
 const colorFondo = "#C7E8C7";

 // Mis 3 postres seleccionados (máximo 3)
 const misPostres = ["A2F4B1", "E4D812", "7A4E9C"];

 // Agregar los postres a la tienda con el ciclo for

   agregarPostre(postre);
 }

──────────────────────────────────────────────────────────────────────
 - Los códigos deben ir entre comillas: "A2F4B1"
 - Separados por comas dentro de los corchetes
 - Selecciona EXACTAMENTE 3 postres
 - El ciclo for debe llamar a: agregarPostre(postre);
 - Los colores deben ser códigos hexadecimales válidos (#RRGGBB)
